-- 1. ตารางข้อมูลโรงเรือน (greenhouses)
CREATE TABLE public.greenhouses (
    greenhouse_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    crop_type VARCHAR(100) NOT NULL,
    area_sqm NUMERIC(8, 2) NOT NULL,
    status VARCHAR(30) DEFAULT 'กำลังใช้งาน',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. ตารางข้อมูลอุปกรณ์และเซนเซอร์ (devices)
CREATE TABLE public.devices (
    device_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    greenhouse_id BIGINT NOT NULL REFERENCES public.greenhouses(greenhouse_id) ON DELETE RESTRICT,
    device_name VARCHAR(100) NOT NULL,
    device_type VARCHAR(20) NOT NULL CHECK (device_type IN ('sensor', 'actuator')),
    gpio_pin VARCHAR(20) NOT NULL,
    installed_date DATE DEFAULT CURRENT_DATE
);

-- 3. ตารางบันทึกค่าเซนเซอร์ (sensor_logs)
CREATE TABLE public.sensor_logs (
    log_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    device_id BIGINT NOT NULL REFERENCES public.devices(device_id) ON DELETE CASCADE,
    temperature NUMERIC(5, 2),
    humidity NUMERIC(5, 2),
    ldr_value INT CHECK (ldr_value BETWEEN 0 AND 4095),
    vr_value INT CHECK (vr_value BETWEEN 0 AND 4095),
    recorded_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. ตารางบันทึกการทำงานของอุปกรณ์ควบคุม (actuator_logs)
CREATE TABLE public.actuator_logs (
    action_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    device_id BIGINT NOT NULL REFERENCES public.devices(device_id) ON DELETE CASCADE,
    state INT NOT NULL CHECK (state IN (0, 1)),
    pwm_brightness INT DEFAULT 0 CHECK (pwm_brightness BETWEEN 0 AND 255),
    control_mode VARCHAR(20) DEFAULT 'MANUAL' CHECK (control_mode IN ('AUTO', 'MANUAL')),
    changed_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. เปิดใช้งาน RLS และอนุญาตการทดสอบ (Lab Mode)
ALTER TABLE public.greenhouses ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.devices ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sensor_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.actuator_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public access greenhouses" ON public.greenhouses FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access devices" ON public.devices FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access sensor_logs" ON public.sensor_logs FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access actuator_logs" ON public.actuator_logs FOR ALL USING (true) WITH CHECK (true);
